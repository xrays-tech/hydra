# 实施计划：Admin-UI 多语言国际化（中/英/法/德 四语切换）

- 日期：2026-08-27
- 状态：待审核（审核通过后启动开发）
- 作者：编码智能体（DeepSeek）
- 对应需求：Admin-UI（`admin-ui/` 纯静态、无构建步骤、Rust `include_dir!` 编译期嵌入）增加多语言国际化改造，支持 中文/English/Français/Deutsch 四语切换

## Goal

1. 全 UI 界面支持 中/英/法/德 四语即时切换（导航、页面标题/描述、CRUD 表格列、表单字段/占位符/提示、按钮、toast、确认框、空状态、登录页等）；
2. 顶栏右上角语言切换器（`<select>`）；选择持久化于 localStorage，刷新后保持；
3. 首次访问按浏览器语言自动探测（`zh*`→中文，`fr*`→法语，`de*`→德语，其余→英文）；
4. API Docs 页：8 个 tag 名 + 53 个端点 `summary`（及详情区 UI 骨架文案）翻译为 4 语；`desc`/响应示例/错误描述/代码示例保留英文（摘要级）；
5. 不改 Rust 静态服务逻辑（`include_dir!` 自动打包新文件）；默认语言（en-US）下现有 Playwright E2E 断言全部保持通过；新增语言切换 E2E 用例。

## 架构

- **无构建步骤约束**：沿用现有"预加载全局"模式（同 `api-docs.js`/`stats.js`），新增 `admin-ui/i18n.js`，在 `index.html` 中**最先**加载（`i18n.js → api-docs.js → stats.js → app.js`）。
- **单一字典**：`const I18N = { en, zh, fr, de }`，点分键（如 `crud.providers.title`）；`t(key, vars)` 查找顺序：当前语言 → `en` 回退 → 原 key 回退（开发期缺失可见），支持 `{var}` 插值。
- **语言状态**：`let LANG`（内存）+ localStorage `hydra-admin-lang`；`setLang(code)` 写存储、设 `<html lang>`（zh→`zh-CN`）、触发 `window.__onLangChanged` 钩子。
- **重渲染**：app.js 注册 `__onLangChanged` → `closeModal()`（避免混合语言表单）+ `applyStaticI18n()`（静态外壳 `[data-i18n*]` 属性）+ `renderNav()` + `go(CURRENT)`（当前页重渲染，CRUD/统计/健康均重新拉取——stats 定时器与 health 自带清理逻辑，安全）。
- **词条化**：CRUD/CUSTOM 配置的 `title/nav/singular/desc`、列 `label`、字段 `label/placeholder/tip`、action `label`、选项 `label` 一律改为 i18n key；渲染点统一 `t()`。示例值类占位符（`sk-…`、`gpt-4`、`acme.com`、证书 PEM 等）不翻译。

function detectLang() {
  try {
    const saved = localStorage.getItem(LANG_STORAGE_KEY);
    if (saved && LANGUAGES.some((l) => l.code === saved)) return saved;
  } catch { /* storage unavailable */ }
  const nav = (navigator.language || "en").toLowerCase();
  if (nav.startsWith("zh")) return "zh";
  if (nav.startsWith("fr")) return "fr";
  if (nav.startsWith("de")) return "de";
  return "en";
}

let LANG = detectLang();

function lookup(dict, key) {
  if (!dict) return undefined;
  // 扁平带点 key（如 "common.brand.sub" 作为字面属性名）优先；
  // 否则按嵌套路径遍历。
  if (key in dict && typeof dict[key] === "string") return dict[key];
  let node = dict;
  for (const part of key.split(".")) {
    if (node === null || typeof node !== "object" || !(part in node)) return undefined;
    node = node[part];
  }
  return typeof node === "string" ? node : undefined;
}

function t(key, vars) {
  const val = lookup(I18N[LANG], key) ?? lookup(I18N.en, key) ?? null;
  const s = val === null ? key : val;
  if (!vars) return s;
  return s.replace(/\{(\w+)\}/g, (m, k) => (vars[k] !== undefined ? String(vars[k]) : m));
}

function setLang(code) {
  if (!LANGUAGES.some((l) => l.code === code)) return;
  LANG = code;
  try { localStorage.setItem(LANG_STORAGE_KEY, code); } catch { /* ignore */ }
  document.documentElement.lang = code === "zh" ? "zh-CN" : code;
  if (typeof window.__onLangChanged === "function") window.__onLangChanged(code);
}

function currentLang() { return LANG; }

/* 静态外壳翻译：替换 [data-i18n]（textContent）、[data-i18n-html]（innerHTML，
 * 字典值可信）、[data-i18n-title]（title）、[data-i18n-placeholder]、
 * [data-i18n-aria-label]。 */
function applyStaticI18n(root = document) {
  for (const node of root.querySelectorAll("[data-i18n]")) node.textContent = t(node.dataset.i18n);
  for (const node of root.querySelectorAll("[data-i18n-html]")) node.innerHTML = t(node.dataset.i18nHtml);
  for (const node of root.querySelectorAll("[data-i18n-title]")) node.title = t(node.dataset.i18nTitle);
  for (const node of root.querySelectorAll("[data-i18n-placeholder]")) node.placeholder = t(node.dataset.i18nPlaceholder);
  for (const node of root.querySelectorAll("[data-i18n-aria-label]")) node.setAttribute("aria-label", t(node.dataset.i18nAriaLabel));
}

document.documentElement.lang = LANG === "zh" ? "zh-CN" : LANG;

 * Hydra admin UI — i18n language switching E2E (2026-08-27 i18n plan).

// @ts-check
const { test, expect } = require('@playwright/test');

const BASE = process.env.HYDRA_BASE || 'http://127.0.0.1:8081';
const TOKEN = process.env.HYDRA_ADMIN_TOKEN || 'dev-admin-token';

async function signIn(page) {
  await page.goto(`${BASE}/admin/`);
  await page.locator('#login-overlay').waitFor({ state: 'visible' });
  await page.fill('#login-token', TOKEN);
  await page.click('#login-btn');
  await expect(page.locator('#login-overlay')).toBeHidden({ timeout: 5000 });
}

test.describe('Hydra admin UI — i18n', () => {
  test('switch to 中文 → nav/title re-render → persisted across reload → back to English', async ({ page }) => {
    await signIn(page);
    // Starts in English (en-US default).
    await expect(page.locator('#nav button.nav-item[data-key="providers"]')).toContainText('Providers');

    // Switch to Chinese.
    await page.selectOption('#lang-select', 'zh');
    await expect(page.locator('#nav button.nav-item[data-key="providers"]')).toContainText('提供方');
    await expect(page.locator('#page-title')).toContainText('提供方');
    await expect(page.locator('#token-status')).toContainText('已认证');

    // Persisted across reload (token is in-memory → sign in again).
    await page.reload();
    await page.locator('#login-overlay').waitFor({ state: 'visible' });
    await page.fill('#login-token', TOKEN);
    await page.click('#login-btn');
    await expect(page.locator('#login-overlay')).toBeHidden({ timeout: 5000 });
    await expect(page.locator('#nav button.nav-item[data-key="providers"]')).toContainText('提供方');

    // Back to English.
    await page.selectOption('#lang-select', 'en');
    await expect(page.locator('#nav button.nav-item[data-key="providers"]')).toContainText('Providers');
  });
});
